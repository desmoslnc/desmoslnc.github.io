<script>
const SUPABASE_URL = "https://khlchbswuzvaziegychy.supabase.co";
const SUPABASE_KEY = "sb_publishable_aepadeUOfhUbT_2WvRUHRQ_2zohD0-v";

const db = window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

const messagesEl = document.getElementById("messages");
const input = document.getElementById("messageInput");
const notice = document.getElementById("notice");
const fileInput = document.getElementById("fileInput");

let currentUser = null;
let currentUsername = "";
let sending = false;
let clearing = false;
let receivedMessages = new Set();
let allMessages = [];

function showNotice(message) {
  notice.textContent = message;
}

function initials(name) {
  return (name || "A").slice(0,1).toUpperCase();
}

async function startChat() {
  try {
    const {data: sessionData, error: sessionError} =
      await db.auth.getSession();

    if (sessionError) throw sessionError;

    if (sessionData.session) {
      currentUser = sessionData.session.user;
    } else {
      const {data, error} = await db.auth.signInAnonymously();
      if (error) throw error;
      currentUser = data.user;
    }

    if (!currentUser) throw new Error("No user session was created.");

    await loadProfile();
    await loadMessages();
    listenForMessages();

  } catch (error) {
    console.error("Chat startup error:", error);
    showNotice("Couldn't connect: " + error.message);

    document.getElementById("empty").innerHTML =
      "<strong>Couldn't connect to Global Chat.</strong><br>" +
      "Please check your connection and Supabase settings.";
  }
}

async function loadProfile() {
  const {data, error} = await db
    .from("profiles")
    .select("username")
    .eq("id", currentUser.id)
    .single();

  if (error) throw new Error("Profile error: " + error.message);

  currentUsername = data.username;
  document.getElementById("usernameInput").value = currentUsername;
}

async function getAttachmentUrl(path) {
  const {data, error} = await db.storage
    .from("chat-uploads")
    .createSignedUrl(path, 3600);

  if (error) {
    console.error("Attachment URL error:", error);
    return null;
  }

  return data.signedUrl;
}

async function renderMessage(message) {
  if (receivedMessages.has(String(message.id))) return;
  receivedMessages.add(String(message.id));

  const empty = document.getElementById("empty");
  if (empty) empty.remove();

  const item = document.createElement("div");
  item.className = "message";
  item.dataset.messageId = String(message.id);

  const avatar = document.createElement("div");
  avatar.className = "message-avatar";
  avatar.textContent = initials(message.username);

  const body = document.createElement("div");
  body.className = "message-body";

  const top = document.createElement("div");
  top.className = "message-top";

  const name = document.createElement("span");
  name.className = "message-name";
  name.textContent = message.username || "Anonymous";

  const time = document.createElement("span");
  time.className = "message-time";
  time.textContent = new Date(message.created_at).toLocaleTimeString([], {
    hour:"numeric",
    minute:"2-digit"
  });

  top.appendChild(name);
  top.appendChild(time);

  const content = document.createElement("div");
  content.className = "message-content";
  content.textContent = message.content || "";

  body.appendChild(top);
  body.appendChild(content);

  if (message.attachment_url) {
    const url = await getAttachmentUrl(message.attachment_url);

    if (url) {
      if (/\.(jpg|jpeg|png|gif|webp)$/i.test(message.attachment_name || "")) {
        const img = document.createElement("img");
        img.src = url;
        img.alt = message.attachment_name || "Uploaded image";
        body.appendChild(img);
      } else {
        const link = document.createElement("a");
        link.href = url;
        link.target = "_blank";
        link.rel = "noopener noreferrer";
        link.textContent = "📎 " + (message.attachment_name || "Download file");
        body.appendChild(link);
      }
    }
  }

  item.appendChild(avatar);
  item.appendChild(body);

  messagesEl.appendChild(item);
  messagesEl.scrollTop = messagesEl.scrollHeight;
}

async function loadMessages() {
  const {data, error} = await db
    .from("messages")
    .select("*")
    .order("created_at", {ascending:true})
    .limit(100);

  if (error) throw new Error("Message history error: " + error.message);

  messagesEl.innerHTML = "";
  receivedMessages.clear();
  allMessages = data || [];

  if (!data.length) {
    const empty = document.createElement("div");
    empty.id = "empty";
    empty.className = "empty-state";
    empty.innerHTML =
      "<strong></strong><br>" +
      "By typing in this chat, you agree the developer does not claim any responsibility for anyone using this during school hours.";
    messagesEl.appendChild(empty);
    return;
  }

  for (const message of data) {
    await renderMessage(message);
  }
}

function listenForMessages() {
  db.channel("global-chat")
    .on("postgres_changes", {
      event:"INSERT",
      schema:"public",
      table:"messages"
    }, async payload => {
      allMessages.push(payload.new);
      await renderMessage(payload.new);
    })
    .on("postgres_changes", {
      event:"DELETE",
      schema:"public",
      table:"messages"
    }, async () => {
      try {
        await loadMessages();
      } catch (error) {
        console.error("Message refresh error:", error);
      }
    })
    .subscribe(state => {
      if (state === "CHANNEL_ERROR") {
        showNotice("Realtime connection failed.");
      }
    });
}

async function clearChat() {
  if (clearing || !currentUser) return;

  clearing = true;
  sending = true;
  document.getElementById("sendButton").disabled = true;

  try {
    showNotice("Clearing chat...");

    const {error} = await db
      .from("messages")
      .delete()
      .not("created_at", "is", null);

    if (error) throw error;

    input.value = "";
    await loadMessages();

    showNotice("Chat cleared for everyone!");
  } catch (error) {
    console.error("Clear chat error:", error);
    showNotice("Couldn't clear chat: " + error.message);
  } finally {
    clearing = false;
    sending = false;
    document.getElementById("sendButton").disabled = false;
  }
}

async function sendMessage(content, attachmentPath = null, attachmentName = null) {
  if (sending || !currentUser) return;
  if (!content.trim() && !attachmentPath) return;

  // /clear is a command, not a chat message.
  if (!attachmentPath && content.trim().toLowerCase() === "/clear") {
    await clearChat();
    return;
  }

  sending = true;
  document.getElementById("sendButton").disabled = true;

  try {
    const {error} = await db.from("messages").insert({
      user_id:currentUser.id,
      username:currentUsername,
      content:content.trim() || "Shared a file",
      attachment_url:attachmentPath,
      attachment_name:attachmentName
    });

    if (error) throw error;

    input.value = "";
    showNotice("");
  } catch (error) {
    console.error("Send error:", error);
    showNotice("Message failed: " + error.message);
  } finally {
    sending = false;
    document.getElementById("sendButton").disabled = false;
  }
}

document.getElementById("sendButton").addEventListener("click", () => {
  sendMessage(input.value);
});

input.addEventListener("keydown", event => {
  if (event.key === "Enter" && !event.shiftKey) {
    event.preventDefault();
    sendMessage(input.value);
  }
});

document.getElementById("uploadButton").addEventListener("click", () => {
  fileInput.click();
});

fileInput.addEventListener("change", async () => {
  const file = fileInput.files[0];
  if (!file) return;

  const allowedTypes = [
    "image/jpeg","image/png","image/gif","image/webp",
    "application/pdf","text/plain"
  ];

  if (!allowedTypes.includes(file.type)) {
    showNotice("That file type isn't allowed.");
    fileInput.value = "";
    return;
  }

  if (file.size > 5 * 1024 * 1024) {
    showNotice("Files must be 5 MB or smaller.");
    fileInput.value = "";
    return;
  }

  try {
    showNotice("Uploading file...");

    const safeName = file.name.replace(/[^a-zA-Z0-9._-]/g, "_");
    const path = currentUser.id + "/" + crypto.randomUUID() + "-" + safeName;

    const {error} = await db.storage
      .from("chat-uploads")
      .upload(path, file, {upsert:false});

    if (error) throw error;

    await sendMessage("", path, file.name);
    showNotice("File uploaded!");
  } catch (error) {
    console.error("Upload error:", error);
    showNotice("Upload failed: " + error.message);
  }

  fileInput.value = "";
});

function toggleSettings() {
  document.getElementById("settingsPanel").classList.toggle("open");
}

document.getElementById("settingsButton").addEventListener("click", toggleSettings);

document.getElementById("closeSettings").addEventListener("click", () => {
  document.getElementById("settingsPanel").classList.remove("open");
});

document.getElementById("saveUsername").addEventListener("click", async () => {
  const username = document.getElementById("usernameInput").value.trim();

  if (username.length < 2 || username.length > 24) {
    showNotice("Username must be between 2 and 24 characters.");
    return;
  }

  try {
    const {error} = await db
      .from("profiles")
      .update({username})
      .eq("id", currentUser.id);

    if (error) throw error;

    currentUsername = username;

    showNotice("Username updated!");
    document.getElementById("settingsPanel").classList.remove("open");
  } catch (error) {
    console.error("Username error:", error);
    showNotice("Couldn't change username: " + error.message);
  }
});

startChat();

setInterval(() => {
  if (currentUser && !sending && !clearing) {
    sendMessage("/clear");
  }
}, 5 * 60 * 1000);
